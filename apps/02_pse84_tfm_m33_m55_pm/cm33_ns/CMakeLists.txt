cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})

# Inject the z_pm out-of-tree TF-M secure partition into the TF-M build.
# Set after find_package(Zephyr) so zephyr_property_target exists, but
# before project() so the TF-M ExternalProject CMake picks it up.
set(Z_PM_PARTITION_DIR ${CMAKE_CURRENT_LIST_DIR}/../tfm_partitions/z_pm)
set_property(TARGET zephyr_property_target
    APPEND PROPERTY TFM_CMAKE_OPTIONS
    -DTFM_EXTRA_MANIFEST_LIST_FILES=${Z_PM_PARTITION_DIR}/manifest_list.yaml
    -DTFM_EXTRA_PARTITION_PATHS=${Z_PM_PARTITION_DIR}
    # Round-7 debug: halt (spin) in tfm_core_panic instead of doing
    # NVIC_SystemReset. Lets a debugger read the exception frame that
    # BusFault_Handler / SecureFault_Handler stashed via EXCEPTION_INFO().
    # Remove for production builds.
    -DCONFIG_TFM_HALT_ON_CORE_PANIC=ON
)

project(cm33_ns_blinki)

# Drop the SoC's default pm_state_set + its PRE_KERNEL_1 SYS_INIT
# (which bus-faults from NS under TF-M when CONFIG_PM=y). We provide
# our own pm_state_set in src/power.c.
get_target_property(zephyr_sources zephyr SOURCES)
list(REMOVE_ITEM zephyr_sources
     ${ZEPHYR_BASE}/soc/infineon/edge/pse84/power.c)
set_property(TARGET zephyr PROPERTY SOURCES ${zephyr_sources})

target_sources(app PRIVATE
    src/main.c
    src/indicator.c
    src/power.c
    src/z_pm_client.c
)

# Pick up psa_manifest/sid.h generated by the TF-M build (provides
# Z_PM_SERVICE_HANDLE for the partition we injected above).
target_include_directories(app PRIVATE
    $<TARGET_PROPERTY:tfm,TFM_BINARY_DIR>/api_ns/interface/include
)
