#-------------------------------------------------------------------------------
# Out-of-tree TF-M secure partition: z_pm.
#
# Picked up automatically by the TF-M build when the Zephyr app sets
# TFM_EXTRA_MANIFEST_LIST_FILES and TFM_EXTRA_PARTITION_PATHS via the
# zephyr_property_target TFM_CMAKE_OPTIONS property.
#
# Library name must follow tfm_psa_rot_partition_* or tfm_app_rot_partition_*
# so the SPE linker script places the partition in the correct section.
#-------------------------------------------------------------------------------

cmake_minimum_required(VERSION 3.21)

add_library(tfm_psa_rot_partition_z_pm STATIC)

add_dependencies(tfm_psa_rot_partition_z_pm manifest_tool)

target_sources(tfm_psa_rot_partition_z_pm
    PRIVATE
        ${CMAKE_CURRENT_LIST_DIR}/z_pm_partition.c
)

# Files generated by tfm_parse_manifest_list.py for this partition.
# output_path in manifest_list.yaml is "partitions/z_pm" which gets prefixed
# with ${CMAKE_BINARY_DIR}/generated/.
target_sources(tfm_psa_rot_partition_z_pm
    PRIVATE
        ${CMAKE_BINARY_DIR}/generated/partitions/z_pm/auto_generated/intermedia_z_pm_partition.c
)
target_sources(tfm_partitions
    INTERFACE
        ${CMAKE_BINARY_DIR}/generated/partitions/z_pm/auto_generated/load_info_z_pm_partition.c
)

target_include_directories(tfm_psa_rot_partition_z_pm
    PRIVATE
        ${CMAKE_BINARY_DIR}/generated/partitions/z_pm
)
target_include_directories(tfm_partitions
    INTERFACE
        ${CMAKE_BINARY_DIR}/generated/partitions/z_pm
)

target_link_libraries(tfm_psa_rot_partition_z_pm
    PRIVATE
        platform_s
        tfm_config
        tfm_sprt
        # NOTE (round 7): ifx_pdl_inc_s intentionally removed. The
        # partition today only exposes Z_PM_OP_PING and does not
        # #include "cy_syspm.h". Re-add this dep when the first
        # non-SRF-wrapped PDL op is added (e.g.
        # Cy_SysPm_SetSysDeepSleepMode for DS-RAM/DS-OFF).
)

target_link_libraries(tfm_partitions
    INTERFACE
        tfm_psa_rot_partition_z_pm
)

target_compile_definitions(tfm_config
    INTERFACE
        TFM_PARTITION_Z_PM
)
