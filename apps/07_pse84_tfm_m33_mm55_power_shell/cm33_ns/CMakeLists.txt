cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})

# Inject the z_pm out-of-tree TF-M secure partition into the TF-M build.
# Set after find_package(Zephyr) so zephyr_property_target exists, but
# before project() so the TF-M ExternalProject CMake picks it up.
set(Z_PM_PARTITION_DIR ${CMAKE_CURRENT_LIST_DIR}/../tfm_partitions/z_pm)
set_property(TARGET zephyr_property_target
    APPEND PROPERTY TFM_CMAKE_OPTIONS
    -DTFM_EXTRA_MANIFEST_LIST_FILES=${Z_PM_PARTITION_DIR}/manifest_list.yaml
    -DTFM_EXTRA_PARTITION_PATHS=${Z_PM_PARTITION_DIR}
    # Halt (spin) in tfm_core_panic instead of doing NVIC_SystemReset,
    # so a debugger can read the exception frame that BusFault_Handler
    # / SecureFault_Handler stashed via EXCEPTION_INFO(). Inherited from
    # project 02. Remove for production builds.
    -DCONFIG_TFM_HALT_ON_CORE_PANIC=ON
)

project(pse84_tfm_m33_mm55_power_shell_cm33ns)

# NOTE: project 02 dropped ${ZEPHYR_BASE}/soc/infineon/edge/pse84/power.c
# from the build because its pm_state_set bus-faults from NS when
# CONFIG_PM=y. This project sets CONFIG_PM=n (shell drives PM directly,
# no Zephyr residency policy), so pm_state_set is never reached and
# the SoC's power.c can stay linked in as-is. No CMake surgery.

target_sources(app PRIVATE
    src/main.c
    src/boot_mcwdt_reset.c
    src/gpio_indicators.c
    src/diag.c
    src/pm_phase_log.c
    src/power_manager.c
    src/shell_cmds.c
    src/cmd_noidle.c
    src/cmd_sleep.c
    src/cmd_deep_sleep.c
    src/z_pm_client.c
)

# Pick up psa_manifest/sid.h generated by the TF-M build (provides
# Z_PM_SERVICE_HANDLE for the partition we injected above).
target_include_directories(app PRIVATE
    $<TARGET_PROPERTY:tfm,TFM_BINARY_DIR>/api_ns/interface/include
)
