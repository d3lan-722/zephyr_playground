567 const struct z_arm_mpu_partition unmap_region = {
568 .start = 0x0,
569 .size = CONFIG_CORTEX_M_NULL_POINTER_EXCEPTION_PAGE_SIZE,
570 #if defined(CONFIG_ARMV8_M_BASELINE) || defined(CONFIG_ARMV8_M_MAINLINE)
571 /* Overlapping region (with any permissions)
572 * will result in fault generation
573 */
574 .attr = K_MEM_PARTITION_P_RO_U_NA,
575 #else
576 /* Explicit no-access policy */
577 .attr = K_MEM_PARTITION_P_NA_U_NA,
578 #endif
579 };
580
581 /* The flash region for null pointer dereferencing detection shall
582 * comply with the regular MPU partition definition restrictions
583 * (size and alignment).
584 */
585 _ARCH_MEM_PARTITION_ALIGN_CHECK(0x0,
586 CONFIG_CORTEX_M_NULL_POINTER_EXCEPTION_PAGE_SIZE);
587
588 #if defined(CONFIG_ARMV8_M_BASELINE) || defined(CONFIG_ARMV8_M_MAINLINE)
589 /* ARMv8-M requires that the area:
590 * 0x0 - CORTEX_M_NULL_POINTER_EXCEPTION_PAGE_SIZE
591 * is not unmapped (belongs to a valid MPU region already).
592 */
593 if ((arm_cmse_mpu_region_get(0x0) == -EINVAL) ||
594 (arm_cmse_mpu_region_get(
595 CONFIG_CORTEX_M_NULL_POINTER_EXCEPTION_PAGE_SIZE - 1)
596 == -EINVAL)) {
597 __ASSERT(0,
598 "Null pointer detection page unmapped\n");
599 }
600 #endif
601
602 if (mpu_configure_region(static_regions_num, &unmap_region) == -EINVAL) {
603
604 __ASSERT(0,
605 "Programming null-pointer detection region failed\n");
606 return -EINVAL;
607 }
608
609 static_regions_num++;
610
611 #endif
612 #endif /* CONFIG_NULL_POINTER_EXCEPTION_DETECTION_MPU */
613
614 /* Coherence check for number of regions in Cortex-M0+, M3, and M4. */
615 #if defined(CONFIG_CPU_CORTEX_M0PLUS) || \
616 defined(CONFIG_CPU_CORTEX_M3) || \
617 defined(CONFIG_CPU_CORTEX_M4)
618 __ASSERT(
619 (MPU->TYPE & MPU_TYPE_DREGION_Msk) >> MPU_TYPE_DREGION_Pos == 8,
620 "Invalid number of MPU regions\n");
621 #endif /* CORTEX_M0PLUS || CPU_CORTEX_M3 || CPU_CORTEX_M4 */
622
623 1 return 0;
624 }
